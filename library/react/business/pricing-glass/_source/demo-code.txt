  const [bgIndex, setBgIndex] = useState(0);

  // Truco maestro: si hacen clic en cualquier sitio que no sea un botón, los lleva a tu web
  const handleGlobalClick = (e: React.MouseEvent) => {
    const target = e.target as HTMLElement;
    // Evitamos abrir la web si están haciendo clic en un botón interactivo del componente
    if (!target.closest('button')) {
      window.open('https://uithefactory.com', '_blank');
    }
  };

  return (
    <div 
      onClick={handleGlobalClick}

            <button
              key={color.name}
              onClick={() => setBgIndex(i)}

              className={`w-6 h-6 rounded-full transition-all duration-300 shadow-[inset_0_2px_4px_rgba(0,0,0,0.5)] ${
